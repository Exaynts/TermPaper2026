' === Классы ===
class User {
  - userId: Long
  - nickname: String
  - email: String
  - passwordHash: String
  - firstName: String
  - lastName: String
  - phoneNumber: String
  - dateOfBirth: Date
  - sex: SexEnum
  - mathLevel: MathLevelEnum
  - avatar: String
  - isActive: Boolean
  - isStaff: Boolean
  + getFullName()
  + checkPassword()
}

class Course {
  - courseId: Long
  - name: String
  - price: BigDecimal
  - discount: Integer
  - rating: Double
  - status: CourseStatusEnum
  - description: Text
  - image: String
  - createdAt: DateTime
  - isDeleted: Boolean
  - deletedAt: DateTime
  + publish()
  + updateRating()
}

class Lesson {
  - lessonId: Long
  - order: Integer
  - name: String
  - text: Text
  - description: Text
  - image: String
  - video: String
  - taskFile: String
}

class Category {
  - categoryId: Long
  - title: String
  - slug: String
}

class SavedCourse {
  - savedAt: DateTime
}

class RecycleBinCourse {
  - deletedAt: DateTime
}

class PurchasedCourse {
  - progress: Integer
  - purchasedAt: DateTime
  - status: PurchaseStatusEnum
}

class LessonProgress {
  - completedAt: DateTime
}

class CourseRating {
  - rating: Integer
  - createdAt: DateTime
  - updatedAt: DateTime
}
