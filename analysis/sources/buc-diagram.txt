  usecase "Ознакомление с каталогом курсов" as BUC1
  usecase "Регистрация / Вход в систему" as BUC2
  usecase "Управление личным профилем" as BUC3
  usecase "Приобретение доступа к курсу" as BUC4
  usecase "Обучение и отслеживание прогресса" as BUC5
  usecase "Управление избранным" as BUC6
  usecase "Восстановление удалённых курсов" as BUC7
  usecase "Управление контентом (CRUD)" as BUC8

}
