' Диаграмма активности: Покупка курса
' Демонстрирует основной и альтернативный потоки

|Пользователь|
start
:Перейти на страницу курса;
:Нажать кнопку "Buy course";

|Система|
if (Пользователь авторизован?) then (Да)
  :Показать модальное окно\nс подтверждением покупки;
  if (Пользователь подтверждает?) then (Да)
    :Проверить, куплен ли курс;
    if (Курс уже куплен?) then (Да)
      :Сообщить: "Курс уже куплен";
      stop
    else (Нет)
      :Проверить статус курса;
      if (Курс опубликован?) then (Да)
        :Создать запись в PurchasedCourse;
        :Создать уведомление в Notification;
        :Отправить email-уведомление;
        :Обновить интерфейс;
        :Сообщить: "Курс куплен успешно";
        stop
      else (Нет)
        :Сообщить: "Курс недоступен для покупки";
        stop
      endif
    endif
  else (Нет - отмена)
    :Закрыть модальное окно;
    stop
  endif
else (Нет)
  :Перенаправить на страницу входа;
  stop
endif

@enduml
