@startuml
' Диаграмма состояний для сущности Course
' Отражает жизненный цикл курса на платформе MathJam

skinparam state {
  BackgroundColor #f0f8ff
  BorderColor #2a5bd7
  FontColor #000000
}

[*] --> Draft : Создание курса

state Draft {
  [*] --> Черновик
  Черновик : Редактирование контента\n(название, цена, описание, уроки)
}

Draft --> Published : Публикация (админ/автор)
Draft --> Archived : Прямая архивация (без публикации)

state Published {
  [*] --> Опубликован
  Опубликован : Доступен для покупки\nОтображается в каталоге\nДоступен для изучения
}

Published --> Draft : Возврат в черновик (редактирование)
Published --> Archived : Архивирование (админ/автор)

state Archived {
  [*] --> Заархивирован
  Заархивирован : Скрыт из каталога\nНедоступен для покупки\nСохранён в базе
}

Archived --> Published : Восстановление из архива

Archived --> [*] : Удаление (полное)
Draft --> [*] : Удаление (полное)
Published --> [*] : Удаление (полное)

@enduml
